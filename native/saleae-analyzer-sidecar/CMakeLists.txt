cmake_minimum_required(VERSION 3.20)
project(sigpilot_saleae_analyzer_sidecar LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
get_filename_component(SIGPILOT_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(SALEAE_RUNTIME_SOURCE "${SIGPILOT_ROOT}/vendor/saleae-standalone/runtime")
set(SALEAE_RUNTIME_BUILD "" CACHE PATH "Standalone runtime build directory")
set(JSON_INCLUDE_DIR "${SIGPILOT_ROOT}/vendor/nlohmann-json/single_include" CACHE PATH "nlohmann/json headers")
set(SIDECAR_OPT_LEVEL "-O3" CACHE STRING "Sidecar optimization flag")
if(NOT EXISTS "${JSON_INCLUDE_DIR}/nlohmann/json.hpp")
    message(FATAL_ERROR "nlohmann/json header not found in ${JSON_INCLUDE_DIR}")
endif()
find_library(SALEAE_COMPAT_LIBRARY NAMES saleae_compat
    PATHS "${SALEAE_RUNTIME_BUILD}/lib" NO_DEFAULT_PATH REQUIRED)
add_executable(sigpilot-saleae-analyzer-sidecar src/standalone_main.cpp)
target_include_directories(sigpilot-saleae-analyzer-sidecar PRIVATE
    "${SALEAE_RUNTIME_SOURCE}/include" "${JSON_INCLUDE_DIR}")
target_compile_options(sigpilot-saleae-analyzer-sidecar PRIVATE "${SIDECAR_OPT_LEVEL}")
target_link_libraries(sigpilot-saleae-analyzer-sidecar PRIVATE "${SALEAE_COMPAT_LIBRARY}" ${CMAKE_DL_LIBS})
if(MINGW)
    target_link_options(sigpilot-saleae-analyzer-sidecar PRIVATE -static-libgcc -static-libstdc++)
elseif(APPLE)
    set_target_properties(sigpilot-saleae-analyzer-sidecar PROPERTIES BUILD_RPATH "@loader_path/runtime")
else()
    set_target_properties(sigpilot-saleae-analyzer-sidecar PROPERTIES BUILD_RPATH "$ORIGIN/runtime")
endif()
